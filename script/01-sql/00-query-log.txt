1.  Find the top 3 products by revenue in each category

-- SUBQUERY VER

SELECT * FROM (
    SELECT
        cat.name as category,
        p.name as product,
        p.price,
        coalesce(sum(oi.quantity * oi.price),0) as revenue,
        ROW_NUMBER() OVER(
            PARTITION BY cat.name
            ORDER BY coalesce(sum(oi.quantity * oi.price),0) DESC
        ) AS rank_in_category
    FROM 
        categories cat
    INNER JOIN
        products p ON p.category_id = cat.id
    LEFT JOIN
        order_items oi ON oi.product_id = p.id
    GROUP BY cat.name, p.name, p.price
    )
WHERE
    rank_in_category <= 3
    ;

-- CTE VER

WITH ranked_products AS (
    SELECT
        cat.name as category,
        p.name as product,
        p.price,
        coalesce(sum(oi.quantity * oi.price),0) as revenue,
        ROW_NUMBER() OVER(
            PARTITION BY cat.name
            ORDER BY coalesce(sum(oi.quantity * oi.price),0) DESC
        ) AS rank_in_category
    FROM 
        categories cat
    INNER JOIN
        products p ON p.category_id = cat.id
    LEFT JOIN
        order_items oi ON oi.product_id = p.id
    GROUP BY cat.name, p.name, p.price
    )
SELECT * FROM
    ranked_products
WHERE

3.  MIN/MAX with Dates - Finding Order Extremes

SELECT
    o.id as order_id,
    o.order_date,
    o.total,
    c.name as customer,
    (SELECT max(total) 
    FROM orders) as max_order_value,
    CASE 
        WHEN o.total = (SELECT max(total) FROM orders)
            THEN 'HIGHEST VALUE'
        WHEN o.total = (SELECT min(total) FROM orders WHERE status = 'completed')
            THEN 'LOWEST COMPLETED'
        ELSE 'NORMAL'
    END as order_classification
FROM
    orders o 
INNER JOIN

        ON o.customer_id = c.id
